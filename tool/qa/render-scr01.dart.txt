import 'dart:io';
import 'dart:ui' as ui;
import 'package:dsapp/app/providers.dart';
import 'package:dsapp/domain_client/models/today_view.dart';
import 'package:dsapp/domain_client/repositories/today_repository.dart';
import 'package:dsapp/features/today/presentation/today_screen.dart';
import 'package:flutter/material.dart';
import 'package:flutter/rendering.dart';
import 'package:flutter_riverpod/flutter_riverpod.dart';
import 'package:flutter_test/flutter_test.dart';
import 'package:mocktail/mocktail.dart';

class _Repo extends Mock implements TodayRepository {}

/// Renders SCR-01 at the reference viewport and writes the QA evidence the
/// screen contract requires. Not a golden test — a rendering harness.
void main() {
  TestWidgetsFlutterBinding.ensureInitialized();

  Future<void> shoot(WidgetTester tester, String name, TodayView view) async {
    final repo = _Repo();
    when(() => repo.forDynamic(any())).thenAnswer((_) async => view);
    await tester.binding.setSurfaceSize(const Size(390, 844));
    await tester.pumpWidget(
      ProviderScope(
        overrides: [todayRepositoryProvider.overrideWithValue(repo)],
        child: MaterialApp(
          theme: ThemeData(useMaterial3: true),
          home: RepaintBoundary(
            key: const Key('shot'),
            child: TodayScreen(
              dynamicId: 'd1',
              onOpenAttention: () {},
              onCheckIn: () {},
              onOpen: (_) {},
            ),
          ),
        ),
      ),
    );
    await tester.pumpAndSettle();

    final el = find.byKey(const Key('shot')).evaluate().first;
    final boundary = el.findRenderObject()! as RenderRepaintBoundary;
    final image = await boundary.toImage(pixelRatio: 3);
    final bytes = await image.toByteData(format: ui.ImageByteFormat.png);
    File(
      '../design/qa/implementation/scr01-$name.png',
    ).writeAsBytesSync(bytes!.buffer.asUint8List());
  }

  final items = [
    const TodayItem(
      occurrenceId: 'o1',
      title: 'Prepare the bedroom',
      state: 'ACTIVE',
      fromDisplayName: 'Morgan',
    ),
    const TodayItem(
      occurrenceId: 'o2',
      title: 'Evening ritual',
      state: 'ACTIVE',
    ),
    const TodayItem(
      occurrenceId: 'o3',
      title: 'Daily check-in',
      state: 'ACTIVE',
    ),
  ];
  final later = [
    const TodayItem(
      occurrenceId: 'o4',
      title: "Read Morgan's note",
      state: 'ACTIVE',
    ),
    const TodayItem(
      occurrenceId: 'o5',
      title: "Lay out tomorrow's clothes",
      state: 'ACTIVE',
    ),
    const TodayItem(
      occurrenceId: 'o6',
      title: 'One honest journal sentence',
      state: 'ACTIVE',
    ),
    const TodayItem(
      occurrenceId: 'o7',
      title: 'Drink water and reset',
      state: 'ACTIVE',
    ),
    const TodayItem(
      occurrenceId: 'o8',
      title: 'Review the weekend plan',
      state: 'ACTIVE',
    ),
  ];

  testWidgets('render default', (t) async {
    await shoot(
      t,
      'default',
      TodayView(
        priorityItems: items,
        laterItems: later,
        totalCount: 8,
        lastConfirmedAt: DateTime(2026, 8, 29, 21, 18),
      ),
    );
  });

  testWidgets('render with response', (t) async {
    await shoot(
      t,
      'with-response',
      TodayView(
        priorityItems: items,
        laterItems: later,
        totalCount: 8,
        lastConfirmedAt: DateTime(2026, 8, 29, 21, 18),
        recentResponse: RecentResponse(
          occurrenceId: 'o1',
          title: 'Prepare the bedroom',
          type: 'ACKNOWLEDGE',
          text: 'I noticed how carefully you held the boundary. Thank you.',
          sentAt: DateTime(2026, 8, 29, 20, 6),
          senderDisplayName: 'Morgan',
        ),
      ),
    );
  });

  testWidgets('render empty', (t) async {
    await shoot(t, 'empty', TodayView(lastConfirmedAt: DateTime(2026, 8, 29)));
  });
}
